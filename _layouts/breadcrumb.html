{%- comment -%} Breadcrumb include Usage: {% include breadcrumb.html %} Behavior: 1) If page.breadcrumb is an array of {title, url} in front matter, use it. 2) Otherwise, auto-generate from page.url segments. {%- endcomment -%}
    {%- assign crumb_items = page.breadcrumb -%} {%- if crumb_items and crumb_items.size > 0 -%} {%- for it in crumb_items -%}
  1. {%- if it.url and forloop.last == false -%} {{ it.title }} {%- else -%} {{ it.title }} {%- endif -%}
    2. {%- endfor -%} {%- else -%} {%- assign path = page.url | split:'/' | reject:' ' | reject:'' -%} {%- assign accum = '' -%}
  2. Home
    3. {%- for seg in path -%} {%- assign is_last = forloop.last -%} {%- assign accum = accum | append:'/' | append:seg -%} {%- assign seg_title = seg | replace:'-',' ' | replace:'_',' ' | capitalize -%} {%- if is_last -%}
  3. {{ page.title | default: seg_title }}
    4. {%- else -%}
  4. {{ seg_title }}
    5. {%- endif -%} {%- endfor -%} {%- endif -%}
